function drawDashboardCharts(data) {
  // Destroy existing charts
  Object.values(dashboardCharts).forEach(function(c) { c.destroy(); });
  dashboardCharts = {};

  // Score Breakdown — doughnut
  var cb = document.getElementById('chartScoreBreakdown');
  if (cb && data.security_score_breakdown) {
    var labels = Object.keys(data.security_score_breakdown);
    var values = Object.values(data.security_score_breakdown);
    dashboardCharts.score = new Chart(cb, {
      type: 'doughnut',
      data: { labels: labels, datasets: [{ data: values, backgroundColor: ['#3b82f6','#10b981','#f59e0b','#e94560','#8b5cf6'] }] },
      options: { responsive: true, maintainAspectRatio: true, plugins: { legend: { position: 'bottom', labels: { font: { size: 10 }, color: '#a0a0a0' } } } }
    });
  }

  // Rules Health — bar
  var rh = document.getElementById('chartRulesHealth');
  if (rh && data.rules_health_breakdown) {
    var labels2 = Object.keys(data.rules_health_breakdown);
    var values2 = Object.values(data.rules_health_breakdown);
    dashboardCharts.rules = new Chart(rh, {
      type: 'bar',
      data: { labels: labels2, datasets: [{ data: values2, backgroundColor: ['#f59e0b','#e94560','#3b82f6','#6b7280','#10b981'] }] },
      options: { responsive: true, maintainAspectRatio: true, plugins: { legend: { display: false } }, scales: { y: { beginAtZero: true } } }
    });
  }

  // Trend — line
  var tr = document.getElementById('chartTrend');
  if (tr && data.trend_data && data.trend_data.length > 1) {
    var tlabels = data.trend_data.map(function(d) { return d.timestamp ? d.timestamp.substring(0, 10) : ''; });
    var tvalues = data.trend_data.map(function(d) { return d.risk_score || 0; });
    dashboardCharts.trend = new Chart(tr, {
      type: 'line',
      data: { labels: tlabels, datasets: [{ data: tvalues, borderColor: '#e94560', backgroundColor: 'rgba(233,69,96,0.1)', fill: true, tension: 0.3 }] },
      options: { responsive: true, maintainAspectRatio: true, plugins: { legend: { display: false } }, scales: { y: { beginAtZero: true } } }
    });
  }
  // After dashboard charts, also load extra visualizations
  setTimeout(function() {
    loadRiskDonut();
    loadServiceTable();
  }, 200);
}
